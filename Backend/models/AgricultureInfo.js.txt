const mongoose = require('mongoose');

const agriSchema = new mongoose.Schema({
  title: { type: String, required: true, trim: true },
  crop: String,
  season: { 
    type: String, 
    enum: ['Kharif','Rabi','Zaid','General'], 
    default: 'General' 
  },
  content: { type: String, required: true },
  category: { 
    type: String, 
    enum: ['Crop info','Seasonal guidance','Crop care','Pest/disease','Irrigation','Subsidy','Other'], 
    default: 'Other' 
  },
  sourceUrl: String,
  isActive: { type: Boolean, default: true },
}, { timestamps: true });

module.exports = mongoose.model('AgricultureInfo', agriSchema);
