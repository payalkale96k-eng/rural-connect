const AgricultureInfo = require('../models/AgricultureInfo');

exports.getAgriculture = async (req, res) => {
  try {
    const { crop, season, category, active, limit = 50 } = req.query;
    const filter = {};
    if (active !== undefined) filter.isActive = active === 'true';
    if (crop) filter.crop = new RegExp(crop, 'i');
    if (season) filter.season = season;
    if (category) filter.category = category;
    
    const items = await AgricultureInfo.find(filter)
      .limit(parseInt(limit))
      .sort({ createdAt: -1 });
    res.json(items);
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

exports.getAgricultureById = async (req, res) => {
  try {
    const item = await AgricultureInfo.findById(req.params.id);
    if (!item) return res.status(404).json({ message: 'Not found' });
    res.json(item);
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

exports.createAgriculture = async (req, res) => {
  try {
    const item = await AgricultureInfo.create(req.body);
    res.status(201).json(item);
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

exports.updateAgriculture = async (req, res) => {
  try {
    const item = await AgricultureInfo.findByIdAndUpdate(req.params.id, req.body, { new: true });
    if (!item) return res.status(404).json({ message: 'Not found' });
    res.json(item);
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

exports.deleteAgriculture = async (req, res) => {
  try {
    const item = await AgricultureInfo.findById(req.params.id);
    if (!item) return res.status(404).json({ message: 'Not found' });
    await item.deleteOne();
    res.json({ message: 'Deleted' });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};
