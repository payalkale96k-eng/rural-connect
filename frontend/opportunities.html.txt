  <script>
(async () => {
  try {
    const items = await fetchOpportunities({active:'true', limit:20});
    const c = document.getElementById('opp-list');
    if (!items || items.length===0) { c.innerHTML = '<p class="text-muted">No opportunities available.</p>'; return; }
    c.innerHTML = items.map(i => `<div class="card mb-3"><div class="card-body"><h5>${escapeHtml(i.title)}</h5><p class="text-muted">Type: ${escapeHtml(i.type)} | Location: ${escapeHtml(i.location)}</p><p>${escapeHtml(i.description)}</p>${i.applicationLink ? `<a href="${escapeHtml(i.applicationLink)}" target="_blank" class="btn btn-sm btn-primary-rural">Apply</a>` : ''}</div></div>`).join('');
  } catch(err) { document.getElementById('opp-list').innerHTML = '<p class="text-danger">Failed to load.</p>'; }
})();
