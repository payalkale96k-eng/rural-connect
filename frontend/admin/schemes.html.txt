  <script>
if (!isAdmin()) window.location.href = '../login.html';

(async () => {
  try {
    const schemes = await fetchSchemes({ limit: 50 });
    const container = document.getElementById('schemes-list');
    if (!schemes || schemes.length === 0) {
      container.innerHTML = '<p class="text-muted">No schemes found.</p>';
      return;
    }
    container.innerHTML = schemes.map(s => `
      <div class="card mb-2">
        <div class="card-body">
          <h6>${escapeHtml(s.name)}</h6>
          <p class="text-muted">${escapeHtml(s.description)}</p>
          <small class="text-muted">Active: ${s.isActive ? 'Yes' : 'No'}</small>
        </div>
      </div>
    `).join('');
  } catch (err) {
    document.getElementById('schemes-list').innerHTML = '<p class="text-danger">Failed to load schemes.</p>';
  }
})();
  </script>
