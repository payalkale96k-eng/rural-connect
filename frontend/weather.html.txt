  <script>
document.getElementById('weather-form').addEventListener('submit', async e => {
  e.preventDefault();
  const loc = document.getElementById('location').value.trim();
  const resDiv = document.getElementById('weather-result');
  resDiv.style.display = 'block';
  resDiv.innerHTML = '<p class="text-muted">Fetching weather...</p>';
  try {
    const w = await fetchWeather(loc);
    resDiv.innerHTML = `<h4>${escapeHtml(w.location)}</h4><p><strong>Temperature:</strong> ${w.temperature} °C</p><p><strong>Condition:</strong> ${escapeHtml(w.condition)}</p><p><strong>Humidity:</strong> ${w.humidity}% | <strong>Wind Speed:</strong> ${w.windSpeed} m/s</p>`;
  } catch(err) { resDiv.innerHTML = '<p class="text-danger">Failed to fetch weather. Ensure backend has OPENWEATHER_API_KEY configured.</p>'; }
});
