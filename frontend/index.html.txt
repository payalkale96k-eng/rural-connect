  <script>
document.addEventListener('DOMContentLoaded', async () => {
  // Load Alerts
  try {
    const alerts = await fetchAlerts({ limit: 3 });
    const ac = document.getElementById('alerts-container');
    if (!alerts || alerts.length === 0) {
      ac.innerHTML = '<p class="text-muted">No emergency alerts at the moment.</p>';
    } else {
      ac.innerHTML = alerts.map(a => `
        <div class="alert-banner ${a.severity === 'Critical' ? 'critical' : ''}">
          <strong>${escapeHtml(a.title)}</strong>
          <p class="mb-0">${escapeHtml(a.message)}</p>
          <small class="text-muted">Location: ${escapeHtml(a.location)}</small>
        </div>
      `).join('');
    }
  } catch (err) {
    document.getElementById('alerts-container').innerHTML = '<p class="text-danger">Failed to load alerts.</p>';
  }

  // Load Schemes
  try {
    const schemes = await fetchSchemes({ limit: 3, active: 'true' });
    const sc = document.getElementById('schemes-container');
    if (!schemes || schemes.length === 0) {
      sc.innerHTML = '<div class="col-12 text-center text-muted">No schemes available.</div>';
    } else {
      sc.innerHTML = schemes.map(s => `
        <div class="col-md-4 mb-4">
          <div class="card h-100">
            <div class="card-body">
              <h5 class="card-title">${escapeHtml(s.name)}</h5>
              <p class="card-text text-muted">${truncateText(s.description, 100)}</p>
              <a href="scheme-detail.html?id=${s._id}" class="btn btn-sm btn-primary-rural">View Details</a>
            </div>
          </div>
        </div>
      `).join('');
    }
  } catch (err) {
    document.getElementById('schemes-container').innerHTML = '<div class="col-12 text-center text-danger">Failed to load schemes.</div>';
  }

  // Load Announcements
  try {
    const anns = await fetchAnnouncements({ limit: 3, active: 'true' });
    const anc = document.getElementById('announcements-container');
    if (!anns || anns.length === 0) {
      anc.innerHTML = '<p class="text-muted">No announcements at the moment.</p>';
    } else {
      anc.innerHTML = anns.map(a => `
        <div class="card mb-3">
          <div class="card-body">
            <h5 class="card-title">${escapeHtml(a.title)}</h5>
            <p class="card-text">${escapeHtml(a.message)}</p>
            <small class="text-muted">Location: ${escapeHtml(a.location)}</small>
          </div>
        </div>
      `).join('');
    }
  } catch (err) {
    document.getElementById('announcements-container').innerHTML = '<p class="text-danger">Failed to load announcements.</p>';
  }

  // Weather Form
  document.getElementById('weather-form').addEventListener('submit', async (e) => {
    e.preventDefault();
    const location = document.getElementById('weather-location').value.trim();
    const resultDiv = document.getElementById('weather-result');
    resultDiv.style.display = 'block';
    resultDiv.innerHTML = '<p class="text-muted">Fetching weather...</p>';

    try {
      const weather = await fetchWeather(location);
      resultDiv.innerHTML = `
        <h5>${escapeHtml(weather.location)}</h5>
        <p><strong>Temperature:</strong> ${weather.temperature} °C</p>
        <p><strong>Condition:</strong> ${escapeHtml(weather.condition)}</p>
        <p><strong>Humidity:</strong> ${weather.humidity}% | <strong>Wind:</strong> ${weather.windSpeed} m/s</p>
      `;
    } catch (err) {
      resultDiv.innerHTML = '<p class="text-danger">Failed to fetch weather. Ensure backend is running with OPENWEATHER_API_KEY.</p>';
    }
  });
});
  </script>
