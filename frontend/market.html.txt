async function loadMarket() {
  const crop = document.getElementById('crop').value;
  const market = document.getElementById('market').value;
  const params = {};
  if (crop) params.crop = crop;
  if (market) params.market = market;
  try {
    const items = await fetchMarket(params);
    const tb = document.getElementById('market-table');
    if (!items || items.length===0) { tb.innerHTML = '<tr><td colspan="6" class="text-center text-muted">No data</td></tr>'; return; }
    tb.innerHTML = items.map(i => `<tr><td>${escapeHtml(i.crop)}</td><td>${escapeHtml(i.market)}</td><td>₹${i.price}</td><td>${escapeHtml(i.unit)}</td><td>${escapeHtml(i.location)}</td><td>${new Date(i.date).toLocaleDateString()}</td></tr>`).join('');
  } catch(err) { document.getElementById('market-table').innerHTML = '<tr><td colspan="6" class="text-danger">Failed</td></tr>'; }
}
document.getElementById('search-btn').addEventListener('click', loadMarket);
loadMarket();
  </script>
