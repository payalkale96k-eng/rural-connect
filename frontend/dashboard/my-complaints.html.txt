  <script>
// Check authentication
if (!isLoggedIn()) {
  window.location.href = '../login.html';
}

(async () => {
  try {
    const complaints = await fetchMyComplaints();
    const container = document.getElementById('complaints-list');
    
    if (!complaints || complaints.length === 0) {
      container.innerHTML = '<p class="text-muted">You have not submitted any complaints yet.</p>';
      return;
    }

    container.innerHTML = complaints.map(comp => `
      <div class="card mb-3">
        <div class="card-body">
          <h5>
            ${escapeHtml(comp.title)} 
            <span class="badge ${getStatusBadge(comp.status)}">${escapeHtml(comp.status)}</span>
          </h5>
          <p class="text-muted">
            Category: ${escapeHtml(comp.category)} | 
            Location: ${escapeHtml(comp.location)}
          </p>
          <p>${escapeHtml(comp.description)}</p>
          <a href="complaint-detail.html?id=${comp._id}" class="btn btn-sm btn-outline-primary">View Details</a>
        </div>
      </div>
    `).join('');
  } catch (err) {
    document.getElementById('complaints-list').innerHTML = '<p class="text-danger">Failed to load complaints.</p>';
  }
})();

function getStatusBadge(status) {
  const badges = {
    'Submitted': 'bg-secondary',
    'Under Review': 'bg-warning',
    'In Progress': 'bg-info',
    'Resolved': 'bg-success'
  };
  return badges[status] || 'bg-secondary';
}
