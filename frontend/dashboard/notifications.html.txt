  <script>
// Check authentication
if (!isLoggedIn()) {
  window.location.href = '../login.html';
}

(async () => {
  try {
    const notifs = await fetchNotifications({ limit: 20 });
    const container = document.getElementById('notifications-list');
    
    if (!notifs || notifs.length === 0) {
      container.innerHTML = '<p class="text-muted">No notifications.</p>';
      return;
    }

    container.innerHTML = notifs.map(n => `
      <div class="card mb-2 ${n.isRead ? '' : 'border-warning'}">
        <div class="card-body">
          <h6>${escapeHtml(n.title)}</h6>
          <p>${escapeHtml(n.message)}</p>
          <small class="text-muted">${new Date(n.createdAt).toLocaleString()}</small>
        </div>
      </div>
    `).join('');
  } catch (err) {
    document.getElementById('notifications-list').innerHTML = '<p class="text-danger">Failed to load notifications.</p>';
  }
})();
  </script>
