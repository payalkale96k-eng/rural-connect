// Check authentication
if (!isLoggedIn()) {

// Load announcements
(async () => {
  try {
    const anns = await fetchAnnouncements({ limit: 5 });
    const container = document.getElementById('announcements');
    if (!anns || anns.length === 0) {
      container.innerHTML = '<p class="text-muted">No announcements.</p>';
    } else {
      container.innerHTML = anns.map(a => `
        <div class="card mb-2">
          <div class="card-body">
            <h6>${escapeHtml(a.title)}</h6>
            <p>${escapeHtml(a.message)}</p>
            <small class="text-muted">${escapeHtml(a.location)}</small>
          </div>
        </div>
      `).join('');
    }
  } catch (err) {
    // Silent fail for announcements
  }
})();
