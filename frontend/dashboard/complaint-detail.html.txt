  <script>
// Check authentication
if (!isLoggedIn()) {
  window.location.href = '../login.html';
}

(async () => {
  const params = new URLSearchParams(window.location.search);
  const id = params.get('id');
  
  if (!id) {
    document.getElementById('complaint-detail').innerHTML = '<p class="text-danger">Invalid complaint ID.</p>';
    return;
  }

  try {
    const comp = await fetchComplaintById(id);
    document.getElementById('complaint-detail').innerHTML = `
      <div class="card">
        <div class="card-body">
          <h4>
            ${escapeHtml(comp.title)} 
            <span class="badge ${getStatusBadge(comp.status)}">${escapeHtml(comp.status)}</span>
          </h4>
          <p><strong>Category:</strong> ${escapeHtml(comp.category)}</p>
          <p><strong>Location:</strong> ${escapeHtml(comp.location)}</p>
          <p>${escapeHtml(comp.description)}</p>
          ${comp.imageUrl ? `<p><strong>Image:</strong> <a href="${escapeHtml(comp.imageUrl)}" target="_blank">View Image</a></p>` : ''}
          ${comp.adminNotes ? `<div class="alert alert-info"><strong>Admin Notes:</strong> ${escapeHtml(comp.adminNotes)}</div>` : ''}
          <p class="text-muted">
            <small>
              Submitted: ${new Date(comp.createdAt).toLocaleString()}<br>
              ${comp.resolvedAt ? `Resolved: ${new Date(comp.resolvedAt).toLocaleString()}` : ''}
            </small>
          </p>
        </div>
      </div>
    `;
  } catch (err) {
    document.getElementById('complaint-detail').innerHTML = '<p class="text-danger">Failed to load complaint details.</p>';
  }
})();

function getStatusBadge(status) {
  const badges = {
    'Submitted': 'bg-secondary',
    'Under Review': 'bg-warning',
    'In Progress': 'bg-info',
    'Resolved': 'bg-success'
  };
  return badges[status] || 'bg-secondary';
}
