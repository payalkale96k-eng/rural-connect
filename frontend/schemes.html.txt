async function loadSchemes() {
  const search = document.getElementById('search').value;
  const category = document.getElementById('category').value;
  const params = { active: 'true' };
  if (search) params.search = search;
  if (category) params.category = category;

  try {
    const schemes = await fetchSchemes(params);
    const container = document.getElementById('schemes-list');
    if (!schemes || schemes.length === 0) {
      container.innerHTML = '<div class="col-12 text-center text-muted">No schemes found.</div>';
      return;
    }
    container.innerHTML = schemes.map(s => `
      <div class="col-md-6 mb-4">
        <div class="card h-100">
          <div class="card-body">
            <h5>${escapeHtml(s.name)}</h5>
            <p class="text-muted">${escapeHtml(s.description)}</p>
            ${s.eligibility ? `<p><strong>Eligibility:</strong> ${escapeHtml(s.eligibility)}</p>` : ''}
            ${s.applicationLink ? `<a href="${escapeHtml(s.applicationLink)}" target="_blank" class="btn btn-sm btn-primary-rural">Apply Now</a>` : ''}
            <a href="scheme-detail.html?id=${s._id}" class="btn btn-sm btn-outline-secondary ms-2">Details</a>
          </div>
        </div>
      </div>
    `).join('');
  } catch (err) {
    document.getElementById('schemes-list').innerHTML = '<div class="col-12 text-danger">Failed to load schemes.</div>';
  }
}

document.getElementById('search-btn').addEventListener('click', loadSchemes);
loadSchemes();
