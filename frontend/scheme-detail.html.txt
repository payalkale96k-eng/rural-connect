  <script src="js/api.js"></script>
  <script>
(async () => {
  const params = new URLSearchParams(window.location.search);
  const id = params.get('id');
  if (!id) { document.getElementById('scheme-detail').innerHTML = '<p>Invalid scheme ID.</p>'; return; }
  try {
    const s = await fetchSchemeById(id);
    document.getElementById('scheme-detail').innerHTML = `
      <div class="card">
        <div class="card-body">
          <h3>${escapeHtml(s.name)}</h3>
          <p><strong>Description:</strong> ${escapeHtml(s.description)}</p>
          ${s.benefits ? `<p><strong>Benefits:</strong> ${escapeHtml(s.benefits)}</p>` : ''}
          ${s.eligibility ? `<p><strong>Eligibility:</strong> ${escapeHtml(s.eligibility)}</p>` : ''}
          ${s.requiredDocuments && s.requiredDocuments.length ? `<p><strong>Documents:</strong> ${s.requiredDocuments.map(d=>escapeHtml(d)).join(', ')}</p>` : ''}
          ${s.applicationLink ? `<a href="${escapeHtml(s.applicationLink)}" target="_blank" class="btn btn-primary-rural">Apply Online</a>` : ''}
        </div>
      </div>
    `;
  } catch(err) { document.getElementById('scheme-detail').innerHTML = '<p class="text-danger">Failed to load scheme.</p>'; }
})();
  </script>
